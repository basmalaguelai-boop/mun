        // Language Toggler Functionality
        const langToggleBtn = document.getElementById('langToggle');
        let currentLang = 'EN';

        langToggleBtn.addEventListener('click', () => {
            if (currentLang === 'EN') {
                currentLang = 'FR';
                langToggleBtn.innerText = 'EN';
                document.querySelectorAll('.content-en').forEach(el => el.classList.add('hidden'));
                document.querySelectorAll('.content-fr').forEach(el => el.classList.remove('hidden'));
            } else {
                currentLang = 'EN';
                langToggleBtn.innerText = 'FR';
                document.querySelectorAll('.content-fr').forEach(el => el.classList.add('hidden'));
                document.querySelectorAll('.content-en').forEach(el => el.classList.remove('hidden'));
            }
        });

        // Carousel Slideshow Script Logic
        const slides = document.querySelectorAll('.carousel-slide');
        const dots = document.querySelectorAll('.carousel-dot');
        const prevBtn = document.getElementById('prevSlide');
        const nextBtn = document.getElementById('nextSlide');
        let currentSlideIdx = 0;
        let slideInterval;

        function updateCarouselSlides() {
            slides.forEach((slide, idx) => {
                if (idx === currentSlideIdx) {
                    slide.classList.remove('opacity-0', 'pointer-events-none');
                    slide.classList.add('opacity-100', 'z-10');
                } else {
                    slide.classList.remove('opacity-100', 'z-10');
                    slide.classList.add('opacity-0', 'pointer-events-none');
                }
            });

            dots.forEach((dot, idx) => {
                if (idx === currentSlideIdx) {
                    dot.classList.remove('bg-white/50');
                    dot.classList.add('bg-gold', 'w-4'); 
                } else {
                    dot.classList.remove('bg-gold', 'w-4');
                    dot.classList.add('bg-white/50');
                }
            });
        }

        function nextCarouselSlide() {
            currentSlideIdx = (currentSlideIdx + 1) % slides.length;
            updateCarouselSlides();
        }

        function prevCarouselSlide() {
            currentSlideIdx = (currentSlideIdx - 1 + slides.length) % slides.length;
            updateCarouselSlides();
        }

        function startAutoSlide() {
            slideInterval = setInterval(nextCarouselSlide, 5000);
        }

        function resetSlideTimer() {
            clearInterval(slideInterval);
            startAutoSlide();
        }

        nextBtn.addEventListener('click', () => { nextCarouselSlide(); resetSlideTimer(); });
        prevBtn.addEventListener('click', () => { prevCarouselSlide(); resetSlideTimer(); });
        dots.forEach(dot => {
            dot.addEventListener('click', (e) => {
                currentSlideIdx = parseInt(e.target.getAttribute('data-slide'));
                updateCarouselSlides();
                resetSlideTimer();
            });
        });

        // Initialize carousel settings on startup
        updateCarouselSlides();
        startAutoSlide();
    </script>
